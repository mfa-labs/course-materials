# BAB 7: Project Akhir – Website Portofolio

---

## 🎯 Tujuan Pembelajaran

Setelah menyelesaikan bab ini, peserta diharapkan mampu:

- Merencanakan dan menyusun struktur folder project web
- Membangun halaman portofolio lengkap dengan Navbar, Hero, About, Projects, dan Contact
- Mengintegrasikan HTML, CSS, dan JavaScript dalam satu project yang kohesif
- Membuat desain yang responsif dan layak ditampilkan sebagai portofolio

---

## 📁 Struktur Folder Project

Sebelum memulai coding, pastikan folder project sudah tersusun rapi:

```
web-portfolio/
├── index.html          ← Halaman utama (satu-satunya file HTML)
├── style.css           ← Semua styling
├── script.js           ← Semua JavaScript
└── images/             ← Semua gambar
    ├── profil.jpg      ← Foto profil
    ├── proyek1.png     ← Screenshot proyek 1
    ├── proyek2.png     ← Screenshot proyek 2
    └── proyek3.png     ← Screenshot proyek 3
```

> 💡 **Tips Best Practice:**
> - Satu file HTML untuk website satu halaman (single-page) — lebih mudah di-deploy
> - Nama file dan folder selalu **huruf kecil** dan tanpa spasi
> - Gambar dioptimalkan sebelum digunakan (gunakan format WebP atau kompres JPG/PNG)

---

## 🎨 Desain Halaman

Website portofolio akan terdiri dari 5 bagian utama:

| Section | Isi |
|---|---|
| **Navbar** | Logo/nama + navigasi yang sticky dan responsif (hamburger di mobile) |
| **Hero** | Nama besar, deskripsi singkat, CTA button |
| **About** | Foto profil + paragraf perkenalan |
| **Projects** | Grid card berisi proyek-proyek |
| **Contact** | Form kontak sederhana + link sosial media |

/* ============================================================

// ===========================
// 1. HAMBURGER MENU
// ===========================
const hamburgerBtn = document.getElementById('hamburgerBtn');
const navbarMenu = document.getElementById('navbarMenu');
const navLinks = document.querySelectorAll('.nav-link');

hamburgerBtn.addEventListener('click', () => {
    hamburgerBtn.classList.toggle('aktif');
    navbarMenu.classList.toggle('aktif');
});

// Tutup menu saat link nav diklik
navLinks.forEach(link => {
    link.addEventListener('click', () => {
        hamburgerBtn.classList.remove('aktif');
        navbarMenu.classList.remove('aktif');
    });
});


// ===========================
// 2. NAVBAR SCROLL EFFECT
// ===========================
const navbar = document.getElementById('navbar');
const scrollTopBtn = document.getElementById('btnScrollTop');

window.addEventListener('scroll', () => {
    // Tambah shadow saat scroll
    if (window.scrollY > 20) {
        navbar.classList.add('scrolled');
    } else {
        navbar.classList.remove('scrolled');
    }

    // Tampilkan tombol scroll to top
    if (window.scrollY > 300) {
        scrollTopBtn.classList.add('terlihat');
    } else {
        scrollTopBtn.classList.remove('terlihat');
    }

    // Highlight nav link aktif
    highlightNavLink();
});


// ===========================
// 3. SMOOTH SCROLL TO TOP
// ===========================
scrollTopBtn.addEventListener('click', () => {
    window.scrollTo({ top: 0, behavior: 'smooth' });
});


// ===========================
// 4. HIGHLIGHT NAV LINK AKTIF
// ===========================
function highlightNavLink() {
    const sections = document.querySelectorAll('section[id]');
    const scrollPosition = window.scrollY + 100;

    sections.forEach(section => {
        const sectionTop = section.offsetTop;
        const sectionHeight = section.offsetHeight;
        const sectionId = section.getAttribute('id');
        const navLink = document.querySelector(`.nav-link[href="#${sectionId}"]`);

        if (navLink) {
            if (scrollPosition >= sectionTop && scrollPosition < sectionTop + sectionHeight) {
                navLinks.forEach(link => link.classList.remove('aktif'));
                navLink.classList.add('aktif');
            }
        }
    });
}


// ===========================
// 5. TYPING EFFECT DI HERO
// ===========================
const typingTexts = [
    "Web Developer 💻",
    "UI/UX Enthusiast 🎨",
    "Lifelong Learner 📚",
    "Problem Solver 🧩"
];

let textIndex = 0;
let charIndex = 0;
let isDeleting = false;
const typingElement = document.getElementById('typingText');

function typeWriter() {
    const currentText = typingTexts[textIndex];

    if (isDeleting) {
        typingElement.textContent = currentText.substring(0, charIndex - 1);
        charIndex--;
    } else {
        typingElement.textContent = currentText.substring(0, charIndex + 1);
        charIndex++;
    }

    let delay = isDeleting ? 60 : 100;

    if (!isDeleting && charIndex === currentText.length) {
        delay = 2000; // Jeda sebelum menghapus
        isDeleting = true;
    } else if (isDeleting && charIndex === 0) {
        isDeleting = false;
        textIndex = (textIndex + 1) % typingTexts.length;
        delay = 400;
    }

    setTimeout(typeWriter, delay);
}

// Mulai typing effect setelah halaman dimuat
document.addEventListener('DOMContentLoaded', () => {
    setTimeout(typeWriter, 500);
});


// ===========================
// 6. ANIMASI FADE-IN SAAT SCROLL (Intersection Observer)
// ===========================
const fadeElements = document.querySelectorAll('.project-card, .about-grid, .contact-grid');

fadeElements.forEach(el => el.classList.add('fade-in'));

const observer = new IntersectionObserver((entries) => {
    entries.forEach(entry => {
        if (entry.isIntersecting) {
            entry.target.classList.add('terlihat');
            observer.unobserve(entry.target); // hentikan observasi setelah terlihat
        }
    });
}, {
    threshold: 0.15
});

fadeElements.forEach(el => observer.observe(el));


// ===========================
// 7. VALIDASI FORM KONTAK
// ===========================
const contactForm = document.getElementById('contactForm');

function tampilkanError(inputId, errorId, pesan) {
    const input = document.getElementById(inputId);
    const errorEl = document.getElementById(errorId);
    input.classList.add('error');
    errorEl.textContent = pesan;
}

function hapusError(inputId, errorId) {
    const input = document.getElementById(inputId);
    const errorEl = document.getElementById(errorId);
    input.classList.remove('error');
    errorEl.textContent = '';
}

function validasiEmail(email) {
    const regex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
    return regex.test(email);
}

contactForm.addEventListener('submit', (e) => {
    e.preventDefault(); // Cegah form terkirim secara default

    const nama = document.getElementById('inputNama').value.trim();
    const email = document.getElementById('inputEmail').value.trim();
    const pesan = document.getElementById('inputPesan').value.trim();
    const formSuccess = document.getElementById('formSuccess');

    let valid = true;

    // Validasi nama
    hapusError('inputNama', 'errorNama');
    if (nama.length < 2) {
        tampilkanError('inputNama', 'errorNama', 'Nama harus minimal 2 karakter.');
        valid = false;
    }

    // Validasi email
    hapusError('inputEmail', 'errorEmail');
    if (!validasiEmail(email)) {
        tampilkanError('inputEmail', 'errorEmail', 'Masukkan alamat email yang valid.');
        valid = false;
    }

    // Validasi pesan
    hapusError('inputPesan', 'errorPesan');
    if (pesan.length < 10) {
        tampilkanError('inputPesan', 'errorPesan', 'Pesan harus minimal 10 karakter.');
        valid = false;
    }

    if (valid) {
        // Simulasi pengiriman (tanpa backend)
        const submitBtn = contactForm.querySelector('button[type="submit"]');
        submitBtn.textContent = 'Mengirim...';
        submitBtn.disabled = true;

        setTimeout(() => {
            contactForm.reset();
            formSuccess.classList.add('tampil');
            submitBtn.textContent = 'Kirim Pesan ✉️';
            submitBtn.disabled = false;

            // Sembunyikan pesan sukses setelah 5 detik
            setTimeout(() => {
                formSuccess.classList.remove('tampil');
            }, 5000);
        }, 1500);
    }
});

// Hapus error saat pengguna mulai mengetik
['inputNama', 'inputEmail', 'inputPesan'].forEach(id => {
    const el = document.getElementById(id);
    el.addEventListener('input', () => {
        el.classList.remove('error');
        document.getElementById('error' + id.replace('input', '')).textContent = '';
    });
});
